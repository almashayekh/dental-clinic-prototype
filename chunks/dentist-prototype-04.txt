function toothPoints(cx,cy,w,h,flip){
  const crown=flip?cy+h*.28:cy-h*.28;
  const neck=flip?cy+h*.05:cy-h*.05;
  const root=flip?cy-h*.48:cy+h*.48;
  const mid=flip?cy-h*.24:cy+h*.24;
  return [
    [cx-w*.42,crown],[cx-w*.50,neck],[cx-w*.30,mid],
    [cx-w*.13,root],[cx,mid],[cx+w*.13,root],
    [cx+w*.30,mid],[cx+w*.50,neck],[cx+w*.42,crown]
  ].map(p=>p.join(",")).join(" ");
}
function buildArch(teeth,isUpper){
  const frag=document.createDocumentFragment(),n=teeth.length;
  teeth.forEach((t,i)=>{
    const u=i/(n-1),x=145+u*610,curve=Math.pow((u-.5)*2,2);
    const y=isUpper?(92+curve*118):(448-curve*118);
    const d=Math.abs(i-7.5),w=d<2?29:d<4?34:39,h=d<2?72:d<4?66:60;
    const angle=(u-.5)*(isUpper?34:-34);
    const g=document.createElementNS("http://www.w3.org/2000/svg","g");
    g.setAttribute("class","tooth-g"+(selected.includes(t)?" selected":""));
    g.setAttribute("tabindex","0");g.setAttribute("role","button");g.setAttribute("aria-label","سن "+t);
    g.setAttribute("transform",`translate(${x} ${y}) rotate(${angle}) translate(${-x} ${-y})`);
    g.onclick=()=>toggleTooth(t);
    g.onkeydown=e=>{if(e.key==="Enter"||e.key===" "){e.preventDefault();toggleTooth(t)}};
    const p=document.createElementNS("http://www.w3.org/2000/svg","polygon");
    p.setAttribute("class","tooth-shape");p.setAttribute("points",toothPoints(x,y,w,h,!isUpper));
    const label=document.createElementNS("http://www.w3.org/2000/svg","text");
    label.setAttribute("class","tooth-label");label.setAttribute("x",x);label.setAttribute("y",y+4);label.textContent=t;
    g.appendChild(p);g.appendChild(label);frag.appendChild(g);
  });
  return frag;
}
